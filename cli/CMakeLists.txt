add_executable(dapple_cli main.cpp film_cmds.cpp film_bench.cpp edit_bench.cpp eval.cpp eval_common.cpp fit_profile.cpp merge_cmds.cpp motion_render.cpp)
set_target_properties(dapple_cli PROPERTIES OUTPUT_NAME dapple)
target_link_libraries(dapple_cli PRIVATE dapple_core dapple)
target_include_directories(dapple_cli PRIVATE ${PROJECT_SOURCE_DIR}/core/src)
if(MSVC)
  target_compile_options(dapple_cli PRIVATE /W4 /utf-8)
else()
  target_compile_options(dapple_cli PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

# `dapple gpu-bench` (core/gpu spike), only when configured with -DDAPPLE_GPU=ON
if(TARGET dapple_gpu)
  target_link_libraries(dapple_cli PRIVATE dapple_gpu)
  target_compile_definitions(dapple_cli PRIVATE DAPPLE_HAVE_GPU=1)
endif()

# `dapple motion render`: edited Motion Photo clips through the node graph (software path, desktop FFmpeg from tools/ffmpeg/build-host.sh)
if(DAPPLE_HAVE_VIDEO_HOST)
  target_link_libraries(dapple_cli PRIVATE dapple_video_host)
  target_include_directories(dapple_cli PRIVATE "${DAPPLE_FFMPEG_HOST_DIR}/include")
  target_compile_definitions(dapple_cli PRIVATE DAPPLE_HAVE_VIDEO_HOST=1)
  if(DAPPLE_FFMPEG_HOST_DLLS)
    add_custom_command(TARGET dapple_cli POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${DAPPLE_FFMPEG_HOST_DLLS} $<TARGET_FILE_DIR:dapple_cli>)
  endif()
endif()
