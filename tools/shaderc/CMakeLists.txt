# dapple_shaderc: host tool (SPIR-V -> ESSL 310 / MSL / manifest). Standalone project so that it can be built for the
# BUILD machine from a cross build (Android) via ExternalProject (see core/cmake/DappleShaderc.cmake).
cmake_minimum_required(VERSION 3.20)
project(dapple_shaderc CXX)
set(CMAKE_CXX_STANDARD 17)
set(DAPPLE_SPIRV_CROSS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../_dl/spirv-cross/SPIRV-Cross-vulkan-sdk-1.4.309.0" CACHE PATH "SPIRV-Cross source")
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_UTIL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SHARED OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_STATIC ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${DAPPLE_SPIRV_CROSS_DIR}" spirv-cross EXCLUDE_FROM_ALL)
add_executable(dapple_shaderc main.cpp)
target_link_libraries(dapple_shaderc PRIVATE spirv-cross-glsl spirv-cross-msl spirv-cross-core)
