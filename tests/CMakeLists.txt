add_executable(dapple_tests core_tests.cpp)
target_link_libraries(dapple_tests PRIVATE dapple_core dapple)
target_include_directories(dapple_tests PRIVATE ${PROJECT_SOURCE_DIR}/core/src)
if(MSVC)
